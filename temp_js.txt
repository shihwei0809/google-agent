async function uploadCoaScreenshot(e) {
        const file = e.target.files[0];
        if (!file) return;

        const formData = new FormData();
        formData.append("file", file);

        document.getElementById("statusBox").innerText = "🖼️ 正在透過瀏覽器 AI 分析 COA 圖片文字座標 (免安裝本地 OCR)... 這可能需要幾秒鐘";
        
        try {
            // 載入 Tesseract.js
            if (typeof Tesseract === 'undefined') {
                const script = document.createElement('script');
                script.src = "https://cdn.jsdelivr.net/npm/tesseract.js@5/dist/tesseract.min.js";
                document.head.appendChild(script);
                await new Promise(resolve => script.onload = resolve);
            }

            const worker = await Tesseract.createWorker('eng');
            const ret = await worker.recognize(file);
            await worker.terminate();

            const imgHeight = ret.data.image_height || (ret.data.lines && ret.data.lines.length > 0 ? ret.data.lines[ret.data.lines.length-1].bbox.y1 + 50 : 1000);
            
            // 0. 抓取目前畫面上已輸入的所有批號，優先作為精確比對標的
            const enteredBatches = [];
            const totalRows = document.getElementById("tableBody") ? document.getElementById("tableBody").children.length : 0;
            for (let i = 1; i <= totalRows; i++) {
                const bEl = document.getElementById(`batch_${i}`);
                if (bEl && bEl.value.trim().length >= 6) {
                    enteredBatches.push(bEl.value.trim().toUpperCase());
                }
            }

            // 1. 抓取所有辨識行與字詞
            let rawBoxes = [];
            let tableHeaderCandidates = [];

            // 尋找各行中的批號或表格特徵
            if (ret.data.lines && ret.data.lines.length > 0) {
                ret.data.lines.forEach(line => {
                    const lText = line.text.trim();
                    const cleanLine = lText.replace(/[^A-Za-z0-9]/g, '').toUpperCase();
                    
                    // 表頭行識別 (包含 Batch, Lot, Item, Spec, Test, Result, No, 批號, 規格, 檢驗, Query Result)
                    if (/batch|lot|item|spec|result|test|date|tank|query\s*result|certificate/i.test(lText) && !/\d{8,}/.test(cleanLine)) {
                        tableHeaderCandidates.push(line.bbox.y1);
                    }

                    // 策略 A: 比對表格中已填寫之批號 (整行去除空白後比對)
                    enteredBatches.forEach(eb => {
                        if (cleanLine.includes(eb)) {
                            rawBoxes.push({
                                text: eb,
                                top: line.bbox.y0,
                                bottom: line.bbox.y1
                            });
                        }
                    });

                    // 策略 B: 正規表達式自動搜尋批號 (例如: 26906E3181 或 8~14 碼包含英文數字)
                    const matches = cleanLine.match(/[0-9]{4,6}[A-Z0-9]{3,8}/g);
                    if (matches) {
                        matches.forEach(m => {
                            const digits = m.replace(/\D/g, '');
                            if (m.length >= 8 && digits.length >= 5) {
                                rawBoxes.push({
                                    text: m,
                                    top: line.bbox.y0,
                                    bottom: line.bbox.y1
                                });
                            }
                        });
                    }

                    // 策略 C: 逐字檢驗 line.words
                    if (line.words && line.words.length > 0) {
                        line.words.forEach(w => {
                            const wText = w.text.replace(/[^A-Za-z0-9]/g, '').toUpperCase();
                            const digits = wText.replace(/\D/g, '');
                            enteredBatches.forEach(eb => {
                                if (wText === eb || (wText.length >= 8 && eb.includes(wText))) {
                                    rawBoxes.push({
                                        text: eb,
                                        top: line.bbox.y0,
                                        bottom: line.bbox.y1
                                    });
                                }
                            });
                            if (wText.length >= 8 && digits.length >= 5) {
                                rawBoxes.push({
                                    text: wText,
                                    top: line.bbox.y0,
                                    bottom: line.bbox.y1
                                });
                            }
                        });
                    }
                });
            }

            // 策略 D: 補充掃描 ret.data.words
            if (ret.data.words && ret.data.words.length > 0) {
                ret.data.words.forEach(w => {
                    const wText = w.text.replace(/[^A-Za-z0-9]/g, '').toUpperCase();
                    const digits = wText.replace(/\D/g, '');
                    enteredBatches.forEach(eb => {
                        if (wText === eb || (wText.length >= 8 && eb.includes(wText))) {
                            rawBoxes.push({
                                text: eb,
                                top: w.bbox.y0,
                                bottom: w.bbox.y1
                            });
                        }
                    });
                    if (wText.length >= 8 && digits.length >= 5) {
                        rawBoxes.push({
                            text: wText,
                            top: w.bbox.y0,
                            bottom: w.bbox.y1
                        });
                    }
                    if (/batch|lot|item|spec|result|test|date|tank|query/i.test(w.text)) {
                        tableHeaderCandidates.push(w.bbox.y1);
                    }
                });
            }

            // 依照垂直 Y 座標由上至下排序
            rawBoxes.sort((a, b) => a.top - b.top);

            // 去除垂直重疊過近的重複框 (同一個批號可能有前後多個單詞)
            let uniqueBoxes = [];
            rawBoxes.forEach(b => {
                const existing = uniqueBoxes.find(ub => Math.abs(ub.top - b.top) < 15);
                if (!existing) {
                    uniqueBoxes.push(b);
                } else if (b.text.length > existing.text.length) {
                    existing.text = b.text;
                }
            });

            // 計算表頭底線 (header_bottom)
            let headerBottom = 0;
            if (uniqueBoxes.length > 0) {
                // 表頭永遠在第一筆批號上方！
                const firstBatchTop = uniqueBoxes[0].top;
                const headersAbove = tableHeaderCandidates.filter(y => y <= firstBatchTop + 5);
                if (headersAbove.length > 0) {
                    headerBottom = Math.max(...headersAbove) + 4;
                } else {
                    headerBottom = Math.max(0, firstBatchTop - 4);
                }
            } else if (tableHeaderCandidates.length > 0) {
                headerBottom = Math.max(...tableHeaderCandidates) + 8;
            } else {
                headerBottom = Math.floor(imgHeight * 0.35);
            }

            // 為每一筆批號分配清晰的 row_top 與 row_bottom 邊界
            let finalBoxes = uniqueBoxes.map((b, idx) => {
                let rowTop = (idx === 0) 
                    ? Math.max(headerBottom, b.top - 8) 
                    : Math.max(headerBottom, Math.floor((uniqueBoxes[idx-1].bottom + b.top) / 2));
                
                let rowBottom = (idx === uniqueBoxes.length - 1)
                    ? b.bottom + 10
                    : Math.floor((b.bottom + uniqueBoxes[idx+1].top) / 2) - 4;

                return {
                    text: b.text,
                    top: b.top,
                    bottom: b.bottom,
                    row_top: rowTop,
                    row_bottom: rowBottom
                };
            });

            formData.append("ocr_data", JSON.stringify({boxes: finalBoxes, header_bottom: headerBottom}));

            const res = await fetch("/api/upload_coa_image", {
                method: "POST",
                body: formData
            });

            if (!res.ok) {
                alert("COA 截圖上傳失敗！");
                return;
            }

            const data = await res.json();
            if (finalBoxes.length > 0) {
                document.getElementById("statusBox").innerText = `✅ 截圖上傳成功！已成功精準定位 ${finalBoxes.length} 個批號數據列。`;
                alert(`🎉 COA 截圖上傳與 AI 辨識成功！\n系統已抓到圖片中 ${finalBoxes.length} 筆批號：\n${finalBoxes.map(b=>'• ' + b.text).join('\n')}\n\n產生 Excel 時會自動為每張單子裁切【表頭 + 該批號專屬數據列】！`);
            } else {
                document.getElementById("statusBox").innerText = "✅ 截圖上傳成功！將由後端 Python 自動分析 TSMC 表格邊界與數據列。";
                alert(`🎉 COA 截圖上傳成功！\n前端文字辨識無直接命中，後端 Python 已啟動【表格幾何邊界識別】，自動為每張單子精確裁切【表頭 + 各筆數據列】！`);
            }
        } catch(err) {
            alert(`COA 截圖處理出錯: ${err}`);
            document.getElementById("statusBox").innerText = "⚠️ 截圖上傳失敗。";
        }
    }
