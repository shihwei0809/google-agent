0269     // ⌨️ 全局鍵盤 Excel 式上下左右與 Enter 導航
0270     function setupKeyboardNavigation() {
0271         const tableBody = document.getElementById("tableBody");
0272         if (!tableBody) return;
0273 
0274         tableBody.addEventListener("keydown", (e) => {
0275             const target = e.target;
0276             if (target.tagName !== "INPUT") return;
0277 
0278             const id = target.id;
0279             if (!id || !id.includes("_")) return;
0280 
0281             const parts = id.split("_");
0282             const colType = parts[0];
0283             const rowIdx = parseInt(parts[1]);
0284             const totalRows = tableBody.children.length;
0285 
0286             const colOrder = ["batch", "loc", "date", "time", "mod"];
0287             const colIndex = colOrder.indexOf(colType);
0288 
0289             // 1. ⬇️ 下鍵 或 ↵ Enter 鍵
0290             if (e.key === "ArrowDown" || e.key === "Enter") {
0291                 e.preventDefault();
0292                 let nextRow = rowIdx + 1;
0293                 if (nextRow > totalRows) {
0294                     addMoreRows(5);
0295                 }
0296                 const nextInput = document.getElementById(`${colType}_${nextRow}`);
0297                 if (nextInput) {
0298                     nextInput.focus();
0299                     if (nextInput.select && nextInput.type !== "date") nextInput.select();
0300                 }
0301             }
0302             // 2. ⬆️ 上鍵
0303             else if (e.key === "ArrowUp") {
0304                 e.preventDefault();
0305                 if (rowIdx > 1) {
0306                     const prevInput = document.getElementById(`${colType}_${rowIdx - 1}`);
0307                     if (prevInput) {
0308                         prevInput.focus();
0309                         if (prevInput.select && prevInput.type !== "date") prevInput.select();
0310                     }
0311                 }
0312             }
0313             // 3. ⬅️ 左鍵
0314             else if (e.key === "ArrowLeft") {
0315                 const isStart = target.type === "date" || (target.selectionStart === 0 && target.selectionEnd === 0);
0316                 if (isStart) {
0317                     if (colIndex > 0) {
0318                         e.preventDefault();
0319                         const leftInput = document.getElementById(`${colOrder[colIndex - 1]}_${rowIdx}`);
0320                         if (leftInput) {
0321                             leftInput.focus();
0322                             if (leftInput.select && leftInput.type !== "date") leftInput.select();
0323                         }
0324                     } else if (rowIdx > 1) {
0325                         e.preventDefault();
0326                         const leftInput = document.getElementById(`${colOrder[colOrder.length - 1]}_${rowIdx - 1}`);
0327                         if (leftInput) {
0328                             leftInput.focus();
0329                             if (leftInput.select && leftInput.type !== "date") leftInput.select();
0330                         }
0331                     }
0332                 }
0333             }
0334             // 4. ➡️ 右鍵
0335             else if (e.key === "ArrowRight") {
0336                 const isEnd = target.type === "date" || (target.selectionEnd === target.value.length);
0337                 if (isEnd) {
0338                     if (colIndex < colOrder.length - 1 && colIndex !== -1) {
0339                         e.preventDefault();
0340                         const rightInput = document.getElementById(`${colOrder[colIndex + 1]}_${rowIdx}`);
0341                         if (rightInput) {
0342                             rightInput.focus();
0343                             if (rightInput.select && rightInput.type !== "date") rightInput.select();
0344                         }
0345                     } else if (rowIdx < totalRows) {
0346                         e.preventDefault();
0347                         const rightInput = document.getElementById(`${colOrder[0]}_${rowIdx + 1}`);
0348                         if (rightInput) {
0349                             rightInput.focus();
0350                             if (rightInput.select && rightInput.type !== "date") rightInput.select();
0351                         }
0352                     }
0353                 }
0354             }
0355         });
0356     }

0646     function onPasteHandler(e, startIdx) {
0647         e.preventDefault();
0648         const clipboardData = e.clipboardData || window.clipboardData;
0649         const pastedData = clipboardData.getData('Text');
0650         if (!pastedData) return;
0651 
0652         const lines = pastedData.split(/\r\n|\n|\r/);
0653         let currentIdx = startIdx;
0654 
0655         lines.forEach(line => {
0656             if (!line.trim()) return;
0657             const items = line.split('\t').map(x => x.trim());
0658 
0659             let batchVal = "";
0660             let locVal = "";
0661             let dateVal = "";
0662             let timeVal = "";
0663             let originVal = "";
0664 
0665             items.forEach(item => {
0666                 const upper = item.toUpperCase().trim();
0667                 // 如果是槽號 (如 E309, T305) 則略過，因為批號輸入時會自動計算
0668                 if (upper.length <= 5 && (upper.startsWith('E') || upper.startsWith('T') || upper.startsWith('P')) && /\d/.test(upper)) {
0669                     return;
0670                 }
0671 
0672                 // 檢查是否為時間 (如 09:00, 0900, 1400, 16:30)
0673                 if ((upper.includes(':') || (/^\d{4}$/.test(upper) && (upper.startsWith('0') || upper.startsWith('1') || upper.startsWith('2')))) && !dateVal.includes(upper) && upper !== batchVal && !timeVal) {
0674                     let normTime = normalizeTimeStr(item);
0675                     if (normTime) {
0676                         timeVal = normTime;
0677                     }
0678                 }
0679                 
0680                 // 檢查是否為日期 (包含 / 或 - 且有數字)
0681                 if ((upper.includes('/') || upper.includes('-')) && /\d/.test(upper) && !dateVal) {
0682                     dateVal = upper;
0683                 } else if (upper.length === 10 && !batchVal && /[A-Z]/.test(upper) && /\d/.test(upper)) {
0684                     batchVal = upper;
0685                 } else if ((SERVER_MAPPING[upper] || upper.includes('18P') || upper.includes('15P') || upper.includes('P3') || upper.length <= 6) && !locVal && upper !== batchVal && !upper.includes('/')) {
0686                     locVal = upper;
0687                 }
0688             });
0689 
0690             if (batchVal || locVal || dateVal || timeVal || originVal) {
0691                 if (currentIdx > document.getElementById("tableBody").children.length) {
0692                     addMoreRows(5);
0693                 }
0694                 if (batchVal) {
0695                     document.getElementById(`batch_${currentIdx}`).value = batchVal;
0696                     onBatchChange(currentIdx);
0697                 }
0698                 if (originVal) {
0699                     if (document.getElementById(`origin_${currentIdx}`)) document.getElementById(`origin_${currentIdx}`).value = originVal;
0700                 }
0701                 if (locVal) {
0702                     document.getElementById(`loc_${currentIdx}`).value = locVal;
0703                     onLocChange(currentIdx);
0704                 }
0705                 if (timeVal) {
0706                     document.getElementById(`time_${currentIdx}`).value = timeVal;
0707                 }
0708                 if (dateVal) {
0709                     // 嘗試轉換日期為 YYYY-MM-DD
0710                     let dStr = dateVal.trim();
0711                     let y = new Date().getFullYear();
0712                     let m = "", d = "";
0713                     let match4 = dStr.match(/(\d{4})[-\/](\d{1,2})[-\/](\d{1,2})/);
0714                     let match2 = dStr.match(/^(\d{1,2})[-\/](\d{1,2})$/);
0715                     if (match4) {
0716                         y = match4[1]; m = match4[2]; d = match4[3];
0717                     } else if (match2) {
0718                         m = match2[1]; d = match2[2];
0719                     }
0720                     if (m && d) {
0721                         m = String(m).padStart(2, '0');
0722                         d = String(d).padStart(2, '0');
0723                         document.getElementById(`date_${currentIdx}`).value = `${y}-${m}-${d}`;
0724                     }
0725                 }
0726                 currentIdx++;
0727             }
0728         });
0729     }

1022     function normalizeDateStr(raw) {
1023         if (!raw) return "";
1024         if (raw instanceof Date) {
1025             let y = raw.getFullYear();
1026             let m = String(raw.getMonth() + 1).padStart(2, '0');
1027             let d = String(raw.getDate()).padStart(2, '0');
1028             return `${y}-${m}-${d}`;
1029         }
1030         let str = String(raw).trim();
1031         if (!str) return "";
1032 
1033         // Excel 序列號 (如 45532)
1034         let num = Number(str);
1035         if (!isNaN(num) && num > 20000 && num < 60000) {
1036             let dateObj = new Date(Math.round((num - 25569) * 86400 * 1000));
1037             let y = dateObj.getFullYear();
1038             let m = String(dateObj.getMonth() + 1).padStart(2, '0');
1039             let d = String(dateObj.getDate()).padStart(2, '0');
1040             return `${y}-${m}-${d}`;
1041         }
1042 
1043         // YYYY/MM/DD, YYYY-MM-DD, YYYY.MM.DD
1044         let m4 = str.match(/^(\d{4})[-\/\.](\d{1,2})[-\/\.](\d{1,2})/);
1045         if (m4) {
1046             return `${m4[1]}-${m4[2].padStart(2, '0')}-${m4[3].padStart(2, '0')}`;
1047         }
1048 
1049         // ROC 115/08/28
1050         let mTW = str.match(/^(\d{2,3})[-\/\.](\d{1,2})[-\/\.](\d{1,2})/);
1051         if (mTW && parseInt(mTW[1]) < 1900) {
1052             let y = parseInt(mTW[1]) + 1911;
1053             return `${y}-${mTW[2].padStart(2, '0')}-${mTW[3].padStart(2, '0')}`;
1054         }
1055 
1056         // MM/DD or M/D (e.g. 8/28)
1057         let m2 = str.match(/^(\d{1,2})[-\/\.](\d{1,2})$/);
1058         if (m2) {
1059             let y = new Date().getFullYear();
1060             return `${y}-${m2[1].padStart(2, '0')}-${m2[2].padStart(2, '0')}`;
1061         }
1062 
1063         // MM/DD/YYYY
1064         let mMDY = str.match(/^(\d{1,2})[-\/\.](\d{1,2})[-\/\.](\d{4})/);
1065         if (mMDY) {
1066             return `${mMDY[3]}-${mMDY[1].padStart(2, '0')}-${mMDY[2].padStart(2, '0')}`;
1067         }
1068 
1069         let jsDate = new Date(str);
1070         if (!isNaN(jsDate.getTime()) && jsDate.getFullYear() > 2000) {
1071             let y = jsDate.getFullYear();
1072             let m = String(jsDate.getMonth() + 1).padStart(2, '0');
1073             let d = String(jsDate.getDate()).padStart(2, '0');
1074             return `${y}-${m}-${d}`;

1080     function normalizeTimeStr(raw) {
1081         if (!raw && raw !== 0) return "";
1082 
1083         // 1. JS Date 物件 (SheetJS cellDates: true 產生的 Date，包含 1899-12-30)
1084         if (raw instanceof Date) {
1085             let h = raw.getHours();
1086             let m = raw.getMinutes();
1087             return `${String(h).padStart(2, '0')}${String(m).padStart(2, '0')}`;
1088         }
1089 
1090         let str = String(raw).trim();
1091         if (!str) return "";
1092 
1093         // 2. Excel 小數時間序列 (如 0.375 -> 0900)
1094         let num = Number(str);
1095         if (!isNaN(num) && num > 0 && num < 1) {
1096             let totalMinutes = Math.round(num * 24 * 60);
1097             let h = Math.floor(totalMinutes / 60) % 24;
1098             let m = totalMinutes % 60;
1099             return `${String(h).padStart(2, '0')}${String(m).padStart(2, '0')}`;
1100         }
1101 
1102         // 3. 符合 HH:MM 或 HH:MM:SS (包含 Sat Dec 30 1899 09:00:00)
1103         let mTime = str.match(/(\d{1,2}):(\d{2})/);
1104         if (mTime) {
1105             let h = String(mTime[1]).padStart(2, '0');
1106             let m = String(mTime[2]).padStart(2, '0');
1107             return `${h}${m}`;
1108         }
1109 
1110         // 4. 純 4 碼數字 (如 "0900", "1400")
1111         if (/^\d{4}$/.test(str)) {
1112             return str;
1113         }
1114 
1115         // 5. 3 碼數字 (如 "900" -> "0900")
1116         if (/^\d{3}$/.test(str)) {
1117             return "0" + str;
1118         }
1119 
1120         return "";
1121     }
