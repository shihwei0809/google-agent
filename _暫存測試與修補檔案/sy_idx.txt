0656     function onPasteHandler(e, startIdx) {
0657         e.preventDefault();
0658         const clipboardData = e.clipboardData || window.clipboardData;
0659         const pastedData = clipboardData.getData('Text');
0660         if (!pastedData) return;
0661 
0662         const lines = pastedData.split(/\r\n|\n|\r/);
0663         let currentIdx = startIdx;
0664 
0665         lines.forEach(line => {
0666             if (!line.trim()) return;
0667             const items = line.split('\t').map(x => x.trim());
0668 
0669             let batchVal = "";
0670             let locVal = "";
0671             let dateVal = "";
0672             let timeVal = "";
0673 
0674             items.forEach(item => {
0675                 const upper = item.toUpperCase().trim();
0676                 // 如果是槽號 (如 E309, T305) 則略過，因為批號輸入時會自動計算
0677                 if (upper.length <= 5 && (upper.startsWith('E') || upper.startsWith('T') || upper.startsWith('P')) && /\d/.test(upper)) {
0678                     return;
0679                 }
0680 
0681                 // 檢查是否為時間 (如 09:00, 0900, 1400, 16:30)
0682                 if ((upper.includes(':') || (/^\d{4}$/.test(upper) && (upper.startsWith('0') || upper.startsWith('1') || upper.startsWith('2')))) && !dateVal.includes(upper) && upper !== batchVal && !timeVal) {
0683                     let normTime = normalizeTimeStr(item);
0684                     if (normTime) {
0685                         timeVal = normTime;
0686                     }
0687                 }
0688                 
0689                 // 檢查是否為日期 (包含 / 或 - 且有數字)
0690                 if ((upper.includes('/') || upper.includes('-')) && /\d/.test(upper) && !dateVal) {
0691                     dateVal = upper;
0692                 } else if ((upper.length === 10 || upper.length === 11) && !batchVal && /[A-Z]/.test(upper) && /\d/.test(upper)) {
0693                     batchVal = upper;
0694                 } else if ((SERVER_MAPPING[upper] || upper.includes('18P') || upper.includes('15P') || upper.includes('P3') || upper.length <= 6) && !locVal && upper !== batchVal && !upper.includes('/')) {
0695                     locVal = upper;
0696                 }
0697             });
0698 
0699             if (batchVal || locVal || dateVal || timeVal) {
0700                 if (currentIdx > document.getElementById("tableBody").children.length) {
0701                     addMoreRows(5);
0702                 }
0703                 if (batchVal) {
0704                     document.getElementById(`batch_${currentIdx}`).value = batchVal;
0705                     onBatchChange(currentIdx);
0706                 }
0707                 if (locVal) {
0708                     document.getElementById(`loc_${currentIdx}`).value = locVal;
0709                     onLocChange(currentIdx);
0710                 }
0711                 if (timeVal) {
0712                     document.getElementById(`time_${currentIdx}`).value = timeVal;
0713                 }
0714                 if (dateVal) {
0715                     // 嘗試轉換日期為 YYYY-MM-DD
0716                     let dStr = dateVal.trim();
0717                     let y = new Date().getFullYear();
0718                     let m = "", d = "";
0719                     let match4 = dStr.match(/(\d{4})[-\/](\d{1,2})[-\/](\d{1,2})/);

0721                     if (match4) {
0722                         y = match4[1]; m = match4[2]; d = match4[3];
0723                     } else if (match2) {
0724                         m = match2[1]; d = match2[2];

0728                         d = String(d).padStart(2, '0');

0730                     }
0731                 }
0732                 currentIdx++;
0733             }
0734         });

1028     function normalizeDateStr(raw) {
1029         if (!raw) return "";
1030         if (raw instanceof Date) {
1031             let y = raw.getFullYear();
1032             let m = String(raw.getMonth() + 1).padStart(2, '0');
1033             let d = String(raw.getDate()).padStart(2, '0');
1034             return `${y}-${m}-${d}`;
1035         }
1036         let str = String(raw).trim();
1037         if (!str) return "";
1038 
1039         // Excel 序列號 (如 45532)
1040         let num = Number(str);
1041         if (!isNaN(num) && num > 20000 && num < 60000) {
1042             let dateObj = new Date(Math.round((num - 25569) * 86400 * 1000));
1043             let y = dateObj.getFullYear();
1044             let m = String(dateObj.getMonth() + 1).padStart(2, '0');
1045             let d = String(dateObj.getDate()).padStart(2, '0');
1046             return `${y}-${m}-${d}`;
1047         }
1048 
1049         // YYYY/MM/DD, YYYY-MM-DD, YYYY.MM.DD
1050         let m4 = str.match(/^(\d{4})[-\/\.](\d{1,2})[-\/\.](\d{1,2})/);
1051         if (m4) {
1052             return `${m4[1]}-${m4[2].padStart(2, '0')}-${m4[3].padStart(2, '0')}`;
1053         }
1054 
1055         // ROC 115/08/28
1056         let mTW = str.match(/^(\d{2,3})[-\/\.](\d{1,2})[-\/\.](\d{1,2})/);
1057         if (mTW && parseInt(mTW[1]) < 1900) {
1058             let y = parseInt(mTW[1]) + 1911;
1059             return `${y}-${mTW[2].padStart(2, '0')}-${mTW[3].padStart(2, '0')}`;
1060         }
1061 
1062         // MM/DD or M/D (e.g. 8/28)
1063         let m2 = str.match(/^(\d{1,2})[-\/\.](\d{1,2})$/);
1064         if (m2) {
1065             let y = new Date().getFullYear();
1066             return `${y}-${m2[1].padStart(2, '0')}-${m2[2].padStart(2, '0')}`;
1067         }
1068 
1069         // MM/DD/YYYY
1070         let mMDY = str.match(/^(\d{1,2})[-\/\.](\d{1,2})[-\/\.](\d{4})/);
1071         if (mMDY) {
1072             return `${mMDY[3]}-${mMDY[1].padStart(2, '0')}-${mMDY[2].padStart(2, '0')}`;
1073         }
1074 
1075         let jsDate = new Date(str);
1076         if (!isNaN(jsDate.getTime()) && jsDate.getFullYear() > 2000) {
1077             let y = jsDate.getFullYear();
1078             let m = String(jsDate.getMonth() + 1).padStart(2, '0');
1079             let d = String(jsDate.getDate()).padStart(2, '0');
1080             return `${y}-${m}-${d}`;
1081         }
1082 
1083         return str;

1086     function normalizeTimeStr(raw) {
1087         if (!raw && raw !== 0) return "";
1088 
1089         // 1. JS Date 物件 (SheetJS cellDates: true 產生的 Date，包含 1899-12-30)
1090         if (raw instanceof Date) {
1091             let h = raw.getHours();
1092             let m = raw.getMinutes();
1093             return `${String(h).padStart(2, '0')}${String(m).padStart(2, '0')}`;
1094         }
1095 
1096         let str = String(raw).trim();
1097         if (!str) return "";
1098 
1099         // 2. Excel 小數時間序列 (如 0.375 -> 0900)
1100         let num = Number(str);
1101         if (!isNaN(num) && num > 0 && num < 1) {
1102             let totalMinutes = Math.round(num * 24 * 60);
1103             let h = Math.floor(totalMinutes / 60) % 24;
1104             let m = totalMinutes % 60;
1105             return `${String(h).padStart(2, '0')}${String(m).padStart(2, '0')}`;
1106         }
1107 
1108         // 3. 符合 HH:MM 或 HH:MM:SS (包含 Sat Dec 30 1899 09:00:00)
1109         let mTime = str.match(/(\d{1,2}):(\d{2})/);
1110         if (mTime) {
1111             let h = String(mTime[1]).padStart(2, '0');
1112             let m = String(mTime[2]).padStart(2, '0');
1113             return `${h}${m}`;
1114         }
1115 
1116         // 4. 純 4 碼數字 (如 "0900", "1400")
1117         if (/^\d{4}$/.test(str)) {
1118             return str;
1119         }
1120 
1121         // 5. 3 碼數字 (如 "900" -> "0900")
1122         if (/^\d{3}$/.test(str)) {
1123             return "0" + str;
1124         }
1125 
1126         return "";
1127     }

1302     function filterByDateMode(mode) {
1303         const today = new Date();
1304         const getFmt = (d) => {
1305             let y = d.getFullYear();
1306             let m = String(d.getMonth() + 1).padStart(2, '0');
1307             let day = String(d.getDate()).padStart(2, '0');
1308             return `${y}-${m}-${day}`;
1309         };
1310 
1311         const d0 = getFmt(today);
1312         let d1Obj = new Date(today); d1Obj.setDate(today.getDate() + 1); const d1 = getFmt(d1Obj);
1313         let d2Obj = new Date(today); d2Obj.setDate(today.getDate() + 2); const d2 = getFmt(d2Obj);
1314 
1315         let filtered = [];
1316         let label = "";
1317 
1318         if (mode === 'd0_d2') {
1319             const targets = [d0, d1, d2];
1320             filtered = allExtractedRecords.filter(r => targets.includes(r.date));
1321             label = `「今天~後天 (${d0} ~ ${d2})」`;
1322         } else if (mode === 'd0') {
1323             filtered = allExtractedRecords.filter(r => r.date === d0);
1324             label = `「今天 (${d0})」`;
1325         } else if (mode === 'd1') {
1326             filtered = allExtractedRecords.filter(r => r.date === d1);
1327             label = `「明天 (${d1})」`;
1328         } else if (mode === 'd2') {
1329             filtered = allExtractedRecords.filter(r => r.date === d2);
1330             label = `「後天 (${d2})」`;
1331         } else {
1332             filtered = [...allExtractedRecords];
1333             label = `「全部日期」`;
1334         }
1335 
1336         if (filtered.length === 0) {
1337             alert(`在所有 ${currentSheetCount} 個分頁中，找不到符合 ${label} 的出貨排程！`);
1338             return;
1339         }
1340 
1341         currentFilteredRecords = filtered;
1342         document.getElementById("importModalTotal").innerText = `${label} 共 ${filtered.length} 筆 (全 ${currentSheetCount} 分頁總計 ${allExtractedRecords.length} 筆)`;
