    async function loadAccounts() {
      try {
        const res = await fetch(API + '?action=getAccounts', { cache: 'no-store' });
        const data = await res.json();
        const t = document.querySelector('#accTable tbody');
        t.innerHTML = '';
        data.forEach(a => {
          let r = a.role === 'admin' ? '管理員' : '檢驗員';
          t.innerHTML += `<tr>
            <td>${a.username}</td>
            <td>${r}</td>
            <td style="width: 180px;">
              <button style="background: #f59e0b; padding: 5px 10px;" onclick="openChangePasswordModal(${a.id}, '${a.username}')">修改密碼</button>
              <button style="background: #ef4444; padding: 5px 10px;" onclick="del(${a.id}, '${a.username}')">刪除</button>
            </td>
          </tr>`;
        });
      } catch (err) { }
    }
