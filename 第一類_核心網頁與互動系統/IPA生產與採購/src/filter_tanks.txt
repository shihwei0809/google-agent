function filterTanksBySelectedProduct(productId) {
    const tankSelect = document.getElementById('schedule-tank');
    const sourceTankSelect = document.getElementById('schedule-source-tank');
    const lineSelect = document.getElementById('schedule-line');
    if (!tankSelect || !lineSelect) return;
    
    const selectedLineId = lineSelect.value;
    const selectedLine = state.lines.find(l => l.id === selectedLineId);
    const factoryId = selectedLine ? selectedLine.factoryId : null;
    
    tankSelect.innerHTML = '';
    if(sourceTankSelect) sourceTankSelect.innerHTML = '<option value="">-- 無 (不消耗原料) --</option>';
    
    if (!factoryId) return;

    // Filter target tanks (成品槽): match product, match factory, normal status, and (shared or dedicated to this line)
    const filteredTanks = state.tanks.filter(t => 
      t.productId === productId && 
      t.status === 'normal' &&
      t.factoryId === factoryId &&
      (!t.dedicatedLineId || t.dedicatedLineId === selectedLineId)
    );
    
    // Filter source tanks (原料槽): match factory, normal status, and (shared or dedicated to this line)
    // Exclude tanks that hold the same product as the target (raw material should be different)
    const sourceTanks = state.tanks.filter(t =>
      t.status === 'normal' &&
      t.factoryId === factoryId &&
      t.productId !== productId &&
      (!t.dedicatedLineId || t.dedicatedLineId === selectedLineId)
    );

    
    if (filteredTanks.length === 0) {
      const opt = document.createElement('option');
      opt.value = '';
      opt.innerText = '⚠️ 無適用此產品的正常儲槽，請先新增儲槽';
      tankSelect.appendChild(opt);
    } else {
      filteredTanks.forEach(t => {
        const opt = document.createElement('option');
        opt.value = t.id;
        opt.innerText = `${t.name} (${t.type === 'tank' ? '固定儲槽' : 'Isotank'}) - 容量剩餘 ${formatNumber(t.capacity - t.currentLevel)} L`;
        tankSelect.appendChild(opt);
      });
    }
    
    if (sourceTankSelect) {
      sourceTanks.forEach(t => {
        const prod = state.products.find(p => p.id === t.productId);
        const pName = prod ? prod.name : '未分配產品';
        const opt = document.createElement('option');
        opt.value = t.id;
        opt.innerText = `${t.name} - ${pName} (剩餘 ${formatNumber(t.capacity - t.currentLevel)} L)`; // wait, it should be currentLevel
        // Let's fix the innerText to match the other one
        opt.innerText = `${t.name} - ${pName} (庫存 ${formatNumber(t.currentLevel)} L)`;
